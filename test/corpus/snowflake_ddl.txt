  COMMENT = 'orders rollup'
  CLUSTER BY (ORDER_ID)
AS
SELECT ORDER_ID, SUM(AMOUNT) AS TOTAL FROM orders GROUP BY ORDER_ID;

ALTER MATERIALIZED VIEW mv_orders SUSPEND;
ALTER MATERIALIZED VIEW mv_orders RENAME TO mv_orders_v2;
DROP MATERIALIZED VIEW IF EXISTS mv_orders_v2;

CREATE SEQUENCE IF NOT EXISTS seq_id START = 1 INCREMENT = 1 NOORDER COMMENT = 'ids';
ALTER SEQUENCE seq_id SET INCREMENT = 5;
ALTER SEQUENCE IF EXISTS seq_id RENAME TO seq_id_v2;
DROP SEQUENCE IF EXISTS seq_id_v2;
